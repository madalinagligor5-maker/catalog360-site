Tablouri unidimensionale (și bidimensionale)

Programa 5.B: tablouri unidimensionale pentru memorarea listelor – declarare, acces la un element

La finalul lecției vei ști

  • declararea și parcurgerea tablourilor
  • indicii de la 0 și depășirea limitelor
  • algoritmii clasici pe tablouri în C++
  • matricele ca tablouri bidimensionale
#include <iostream>
using namespace std;

int main() {
    int v[100];                  // dimensiunea maximă se fixează la compilare
    int n = 6;
    int date[] = {5, 2, 9, 1, 7, 3};
    for (int i = 0; i < n; i++) v[i] = date[i];

    int mx = v[0], poz = 0;
    for (int i = 1; i < n; i++)
        if (v[i] > mx) { mx = v[i]; poz = i; }
    cout << "maxim " << mx << " pe pozitia " << poz << "\n";

    for (int i = 0; i < n - 1; i++)        // sortare prin selecție
        for (int j = i + 1; j < n; j++)
            if (v[j] < v[i]) { int aux = v[i]; v[i] = v[j]; v[j] = aux; }
    for (int i = 0; i < n; i++) cout << v[i] << " ";
    cout << "\n";

    int a[3][3] = {{1, 2, 3}, {4, 5, 6}, {7, 8, 9}};
    int sd = 0;
    for (int i = 0; i < 3; i++) sd += a[i][i];
    cout << "diagonala: " << sd << "\n";
    return 0;
}

  • Indicii sunt de la 0 la n − 1. C++ nu verifică depășirea: v[100] într-un tablou de 100 de elemente este comportament nedefinit (poate „merge”, poate strica alte variabile).
  • Tablourile globale sunt inițializate cu 0; cele locale nu (conțin valori reziduale) – inițializează-le: int f[10] = {0};
  • Un tablou transmis unei funcții se transmite prin adresă: modificările din funcție se văd în apelant.
  • Pentru dimensiune variabilă se folosește vector<int> (din <vector>), dar programa cere tablourile clasice.

Aplicația 3.1Inserare și ștergere într-un tabloumediu

Citește n și n numere. Inserează valoarea 0 pe poziția 2 (deplasând elementele spre dreapta), apoi șterge primul element negativ (deplasând spre stânga). Afișează tabloul după fiecare operație.

#include <iostream>
using namespace std;

int main() {
    int v[101], n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> v[i];
    for (int i = n; i > 2; i--) v[i] = v[i - 1];   // de la dreapta la stânga!

    cout << "\n";

    for (int i = 0; i < n && p == -1; i++) if (v[i] < 0) p = i;
    if (p != -1) {
        for (int i = p; i < n - 1; i++) v[i] = v[i + 1];
        n--;
    }
    for (int i = 0; i < n; i++) cout << v[i] << " ";
    cout << "\n";
    return 0;
}

Aplicația 3.2Căutare binară în C++mediu

Citește n, un tablou sortat crescător și o valoare x. Afișează poziția lui x (căutare binară) sau -1.

#include <iostream>
using namespace std;

int main() {
    int v[1000], n, x;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> v[i];
    cin >> x;
    int st = 0, dr = n - 1, poz = -1;
    while (st <= dr && poz == -1) {
        int m = (st + dr) / 2;
        if (v[m] == x) poz = m;
        else if (v[m] < x) st = m + 1;
        else dr = m - 1;
    }
    cout << poz << "\n";
    return 0;
}

Subprograme: funcții, parametri prin valoare și prin referință, recursivitate

Programa 5.B: subprograme, subprograme recursive

La finalul lecției vei ști

  • definirea, declararea (prototipul) și apelul funcțiilor
  • transmiterea prin valoare vs. prin referință (&)
  • funcțiile void și cele care întorc o valoare
  • recursivitatea în C++
#include <iostream>
using namespace std;

int sumaCifre(int n) {               // întoarce o valoare
    int s = 0;
    while (n) { s += n % 10; n /= 10; }
    return s;
}

void dubleaza(int x, int &y) {       // x prin valoare, y prin REFERINȚĂ
    x *= 2;
    y *= 2;
}

long long fact(int n) {              // recursivă
    if (n <= 1) return 1;
    return n * fact(n - 1);
}

void afisVector(int v[], int n) {    // tabloul se transmite prin adresă
    for (int i = 0; i < n; i++) cout << v[i] << " ";
    cout << "\n";
}

int main() {
    int a = 5, b = 5;
    dubleaza(a, b);
    cout << sumaCifre(2026) << " " << a << " " << b << " " << fact(10) << "\n";
    int v[] = {3, 1, 4};
    afisVector(v, 3);
    return 0;
}

  • O funcție trebuie declarată înainte de apel: fie definită mai sus, fie printr-un prototip (int sumaCifre(int n);) înaintea lui main.
  • Prin valoare: funcția lucrează pe o copie. Prin referință (int &y): numele y este un alt nume pentru variabila din apel – modificările rămân.
  • Variabilele locale dispar la ieșirea din funcție; cele globale sunt vizibile peste tot (și inițializate cu 0).

Aplicația 4.1Funcție cmmdc și simplificarea unei fracțiimediu

Scrie funcția int cmmdc(int a, int b) (Euclid) și funcția void simplifica(int &a, int &b). Programul citește o fracție a/b și afișează forma ireductibilă.

Vezi rezolvarea-model
#include <iostream>
using namespace std;

int cmmdc(int a, int b) {
    while (b) { int r = a % b; a = b; b = r; }
    return a;
}

void simplifica(int &a, int &b) {
    int d = cmmdc(a, b);
    a /= d; b /= d;
}

int main() {
    int a, b;
    cin >> a >> b;
    simplifica(a, b);
    cout << a << "/" << b << "\n";
    return 0;
}

Pointeri și alocarea dinamică a memoriei

Programa 5.B: pointeri – declarare, alocare și eliberare dinamică; operatorii & și *, atribuiri, adunarea cu un întreg, scăderea a două adrese

La finalul lecției vei ști

  • ce este o adresă și un pointer
  • operatorii & (adresă) și * (dereferențiere)
  • aritmetica pointerilor pe tablouri

#include <iostream>
#include <string>
using namespace std;

int main() {
    int x = 10;
    int *p = &x;                 // p memorează adresa lui x
    *p = 25;                     // modificăm x prin p
    cout << x << " " << *p << "\n";

    int v[] = {10, 20, 30, 40, 50};
    int *a = v;                  // v este adresa primului element
    int *b = &v[3];
    cout << *(a + 2) << " " << b - a << " " << *b + 1 << " " << *(b + 1) << "\n";
    a++;                         // a indică acum v[1]
    cout << *a << " " << a[1] << "\n";   // a[1] este *(a + 1)

    int n = 4;
    int *t = new int[n];         // tablou alocat dinamic
    for (int i = 0; i < n; i++) t[i] = i * i;
    cout << t[3] << "\n";
    delete[] t;                  // eliberarea memoriei
    t = nullptr;                 // evităm folosirea unei adrese eliberate

    int *q = new int(7);         // un singur întreg
    cout << *q << "\n";
    delete q;
    return 0;

#include <iostream>
#include <string>
using namespace std;

class Cont {
private:                          // accesibil doar din metodele clasei

    double sold;

        cout << "\n";

    cout << "\nfinal main\n";
    return 0;

#include <iostream>
#include <string>
using namespace std;

class Figura {
protected:
    string nume;
public:

int main() {

    cout << "\n";

    cout << "\n";
    return 0;
